event = {
	id = 2100
	random = no
	persistent = yes
	
	name = "Occupation of USSR"
	desc = "As our forces advance deeper into the Soviet Union, we must divide the responsibility of occupation with our allies."
	style = 2
	picture = "news_paper"

	trigger = {	
		OR = {
			country = ROM
			country = HUN
		}
		alliance = { country = GER }
		OR = {
			region = { region = "Poland"		data = 1 value = 1 }
			region = { region = "Belarus"		data = 1 value = 1 }
			region = { region = "Ukraine"		data = 1 value = 1 }
		}
		NOT = {
			AND = {
				country = ROM
				NOT = { region = { region = "Poland"		data = 1 value = 1  } }
				NOT = { region = { region = "Belarus"		data = 1 value = 1  } }
				NOT = { region = { region = "Ukraine"		data = 1 value = 13 } } #<8/58
				control = { province = 233  }
				control = { province = 1565 }
				control = { province = 3404 }
				control = { province = 3405 }
				control = { province = 232  }
				control = { province = 1571 }
				control = { province = 231  }
			}
		}
	}

    date = { day = 0 month = january year = 1936 }
    offset = 30
	deathdate = { day = 29 month = december year = 1963 }

	action = {
		name = "Divide the lands with our allies"
		command = { type = secederegion		which = GER value = "Poland"		when = 1 }
		command = { type = secederegion		which = GER value = "Belarus"		when = 1 }
		command = { type = secederegion		which = GER value = "Ukraine"		when = 1 }
	}
}

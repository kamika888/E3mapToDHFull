event = {  #General Mob Group A
 
	id = 8300066
	random = no
	persistent = yes

	decision = {
		flag = theatsAssessed
		local_flag = AllInit
		NOT = { local_flag = demob }
		#Can jump straight to general mobilization
		#policy = { type = professional_army value = 5 } 
		policy = { type = professional_army value = 1 }
		NOT = { policy = { type = professional_army value = 6 } }
		atwar = yes
		NOT = { lost_national = { value = 100 } }
	}

	decision_trigger = { 
		NOT = { local_flag = demilitarized }
		#only independent states and very strong puppets can do general mobilization
		OR = {
			local_flag = puppetLevel7
			local_flag = puppetLevel8
			local_flag = puppetLevel9
			local_flag = legitIndie
		}
		
		OR = {
			lost_national = { value = 15 }
			AND = {
				lost_national = { value = 3 }
				NOT = { dissent = 10 }
			}
			local_flag = inMajorWar
			policy = { type = defense_lobby value = 7 }
		}
	}

	trigger = {
		NOT = { local_flag = demilitarized }
		flag = theatsAssessed
		local_flag = AllInit
		NOT = { local_flag = demob }
		#Can jump straight to general mobilization
		#policy = { type = professional_army value = 5 } 
		policy = { type = professional_army value = 1 }
		NOT = { policy = { type = professional_army value = 6 } }
		atwar = yes
		
		#NOT = { lost_national = { value = 100 } }
		
		NOT = { #too little to mobilize
			lost_national = { value = 90 }
			AND = {
				country = CHC
				NOT = { region = { region = "China" country = CHC value = 10 } }
			}
		} 
		
		#only independent states and very strong puppets can mobilize
		OR = {
			local_flag = puppetLevel7
			local_flag = puppetLevel8
			local_flag = puppetLevel9
			local_flag = legitIndie
		}
		
		#when general mob happens, the penalties for taking people out of the workforce are immediate, but the manpower still "sits" in the recruitment pool and does nothing until new divisions are actually fielded.
		#In effect, these people are neither working nor fighting, which is unrealistic.
		#all things considered, it is better for "general mobilization" not to be invoked until more manpower is actually needed, otherwise the country is just wasting its IC/resources
		#downside - if your territory is being overrun, failing to enact "general mob" early may cause you to lose out on potential manpower. But the ability to do that itself is unrealistic.
		OR = {
			NOT = { manpower = 70 }
			lost_national = { value = 5 }
			#local_flag = inMajorWar
			AND = {
				under_attack = -1
				OR = {
					country = POL 
					country = VIC
					country = CZE
					AND = {
						country = SOV
						war = { country = GER }
					}
				}
			}
		}
		
		OR = {
			lost_national = { value = 15 }
			AND = {
				lost_national = { value = 3 }
				NOT = { dissent = 10 }
			}
			local_flag = inMajorWar
			#Great Power at war with another Great Power
			AND = {
				OR = { 
					country = GER
					country = ITA
					country = ENG
					country = VIC
					country = SOV
					country = RUS
					country = USA
					country = JAP
					country = CHI
				}
				OR = { 
					war = { country = GER }
					war = { country = ITA }
					war = { country = ENG }
					war = { country = VIC }
					war = { country = SOV }
					war = { country = RUS }
					war = { country = USA }
					war = { country = JAP }
					war = { country = CHI }
				}
			}
			policy = { type = defense_lobby value = 7 }
		}
	}
   
	name = "MOB6name"
	desc = "8300066desc" 
	decision_desc = "DEC_8300066_DESC"
	style = 2
	picture = "Wehrmacht"
	decision_picture = "decision_german_total_war"
	
	date = { day = 1 month = january year = 1914 }
	offset = 15
	deathdate = { day = 30 month = december year = 1963 }

	action = {
